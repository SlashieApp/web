# Typography

The SDL type scale. Every role below is **read live from `sdlTypeScale`** in
`src/theme/styles.ts`, so this page can never drift from the code. Components
should reference these roles rather than hardcoding `fontSize` / `lineHeight` /
`fontWeight` / `letterSpacing` values.

**Font families** resolve from the Chakra theme (`src/theme/chakraSystem.ts`),
from the `--font-geist` and `--font-geist-mono` variables that next/font sets on
`<html>`:

- **Display, headings, body, label & UI text** use Geist (`display`, `heading`
  and `body` are one family).
- **Mono** uses Geist Mono for codes and IDs.

Every heading is **500** with tight tracking, and nothing is heavier than 500.
Prices, counts and other figures are **600 with tabular figures** (`sdlFigure` in
`src/theme/styles.ts`). The Chakra `Heading` takes weight 500 and tracking by
size: up to 16px none, 18px -0.01em, 20px -0.015em, 24px -0.02em, 28px -0.025em,
34 to 36px -0.03em, 44px and up -0.035em.

All specimens render with SDL semantic color roles (`text.default`, `text.muted`),
so they reflect the active light/dark theme.

export const ROLE_META = {
  'display-2xl': { family: 'display', use: 'Hero headline. 44/46 on a phone.' },
  'display-xl': { family: 'display', use: 'Marketing display.' },
  'display-lg': { family: 'display', use: 'Marketing section title.' },
  'heading-xl': { family: 'heading', use: 'Page title, the page header’s big title (H1).' },
  'heading-lg': { family: 'heading', use: 'Section heading (H2).' },
  'heading-md': { family: 'heading', use: 'Subsection / card title (H3).' },
  'heading-sm': { family: 'heading', use: 'Small title, the page header’s bar title.' },
  'text-lg': { family: 'body', use: 'Lead paragraph / emphasis body.' },
  'text-md': { family: 'body', use: 'Default body copy.' },
  'text-sm': { family: 'body', use: 'Secondary body / dense rows.' },
  'text-xs': { family: 'body', use: 'Captions, metadata, footnotes.' },
  'label-md': { family: 'body', use: 'Form labels, buttons, tabs.' },
  'label-sm': { family: 'body', use: 'Small labels.' },
  eyebrow: { family: 'body', use: 'Quiet uppercase label above a heading, in text.subtle.' },
  'mono-sm': { family: 'mono', use: 'Codes, IDs, numeric tokens.' },
}

export const SAMPLE = 'The quick brown Slashie jumps over 0123456789'

export const Specimen = ({ role }) => {
  const scale = sdlTypeScale[role]
  const meta = ROLE_META[role]
  return (
    <Box
      borderBottomWidth="1px"
      borderColor="border.default"
      py={4}
    >
      <Flex
        gap={3}
        justify="space-between"
        align="baseline"
        flexWrap="wrap"
        mb={2}
      >
        <Text
          fontFamily="mono"
          fontSize="12px"
          fontWeight={500}
          color="text.default"
        >
          {role}
        </Text>
        <Text fontFamily="mono" fontSize="12px" color="text.muted">
          {scale.fontSize} / {scale.lineHeight} / {scale.fontWeight} / {scale.letterSpacing}
          {scale.phone ? ` (phone ${scale.phone.fontSize} / ${scale.phone.lineHeight})` : ''} · {meta.family}
        </Text>
      </Flex>
      <Text fontSize="12px" color="text.subtle" mb={2}>
        {meta.use}
      </Text>
      <Text
        fontFamily={meta.family}
        fontSize={scale.fontSize}
        lineHeight={scale.lineHeight}
        fontWeight={scale.fontWeight}
        letterSpacing={scale.letterSpacing}
        textTransform={scale.textTransform}
        color={role === 'eyebrow' ? 'text.subtle' : 'text.default'}
      >
        {SAMPLE}
      </Text>
    </Box>
  )
}

export const TypeScale = () => (
  <Stack gap={0}>
    {Object.keys(sdlTypeScale).map((role) => (
      <Specimen key={role} role={role} />
    ))}
  </Stack>
)
